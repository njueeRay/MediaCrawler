## 0. 与现有数据库层的集成方式

### 0.1 现有架构分析

MediaCrawler 已有完整的数据库基础设施：

| 文件 | 作用 | WebUI 集成方式 |
|------|------|----------------|
| `database/models.py` | 24 个爬虫 ORM 模型，使用 `Base = declarative_base()` | ✅ **不修改**，WebUI 以只读方式查询 |
| `database/db_session.py` | 异步引擎工厂 `get_async_engine()` + 会话管理 `get_session()` | ✅ **直接复用**，新服务用同一个 Session |
| `database/db.py` | `init_db()` → `create_tables()` → `Base.metadata.create_all` | ✏️ **小幅修改**，import WebUI 模型 |
| 无迁移框架 | 没有 Alembic，靠 `create_all` 自动建表 | ✅ **兼容**，新表也由 `create_all` 创建 |

### 0.2 核心策略：共享 Base，独立文件

```python
# database/models.py (现有 - 不修改)
Base = declarative_base()     # ← 24 个爬虫模型共用此 Base

# database/webui_models.py (新建)
from database.models import Base  # ← 导入同一个 Base 实例
#
# 由于共享 Base.metadata，当 create_all() 执行时
# 会同时创建 models.py + webui_models.py 中的所有表
```

### 0.3 为什么不修改 models.py

- `database/models.py` 有 494 行代码、24 个 ORM 模型，是爬虫系统的核心
- 修改已有文件有引入 bug 的风险，且与上游 (NanmiCoder/MediaCrawler) 产生 diff 冲突
- WebUI 模型在逻辑上完全独立，使用 `webui_` 表前缀物理隔离
- 共享同一个 `Base` 后，`create_all()` 能自动覆盖所有表 —— 不需要修改建表逻辑

### 0.4 db.py 唯一改动

```python
# database/db.py — 仅新增一行 import
async def init_db(db_type: str = None):
    import database.webui_models  # 🆕 确保 WebUI 模型注册到 Base.metadata
    await init_table_schema(db_type)
```

### 0.5 新服务如何使用数据库

```python
# 所有新服务统一使用现有的 get_session()
from database.db_session import get_session
from database.webui_models import Subscription

async def list_subscriptions():
    async with get_session() as session:
        if session is None:
            raise HTTPException(400, "数据库未配置 (当前存储模式为 CSV/JSON)")
        result = await session.execute(select(Subscription))
        return result.scalars().all()

# 读取爬虫数据时，使用现有模型
from database.models import XhsNote, WechatArticle

async def query_xhs_notes(page, size):
    async with get_session() as session:
        result = await session.execute(
            select(XhsNote).order_by(XhsNote.add_ts.desc())
            .offset((page-1)*size).limit(size)
        )
        return result.scalars().all()
```

### 2.1 Subscription — 创作者订阅

```python
class Subscription(Base):
    """订阅的创作者"""
    __tablename__ = "webui_subscription"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    platform        = Column(String(20), nullable=False, index=True)
    # 平台: xhs | wechat | dy | bili | wb | ks | tieba | zhihu

    creator_id      = Column(String(128), nullable=False)
    # 平台特定ID: XHS=user_id, WeChat=fakeid, DY=sec_user_id, etc.

    creator_name    = Column(String(256), nullable=False)
    # 显示名称

    creator_avatar  = Column(String(512), default="")
    # 头像 URL

    creator_url     = Column(String(512), default="")
    # 主页链接

    creator_meta    = Column(JSON, default=dict)
    # 额外元数据 (粉丝数、简介、认证等), 结构因平台而异
    # XHS:    {"fans_count": 12500, "note_count": 342, "signature": "..."}
    # WeChat: {"alias": "...", "service_type": "...", "identity_name": "..."}
    # DY:     {"follower_count": 50000, "aweme_count": 128}

    is_active       = Column(Boolean, default=True)
    # 是否活跃订阅

    auto_crawl      = Column(Boolean, default=True)
    # 是否参与自动采集任务

    crawl_config    = Column(JSON, default=dict)
    # 采集配置覆盖: {"max_pages": 5, "enable_comments": false}

    last_crawled_at = Column(DateTime, nullable=True)
    # 最近一次采集时间

    last_content_at = Column(DateTime, nullable=True)
    # 最近一条内容的发布时间

    content_count   = Column(Integer, default=0)
    # 已采集的内容总数

    tags            = Column(JSON, default=list)
    # 用户自定义标签: ["设计", "重要"]

    notes           = Column(Text, default="")
    # 用户备注

    created_at      = Column(DateTime, default=func.now())
    updated_at      = Column(DateTime, default=func.now(), onupdate=func.now())

    # 联合唯一约束: 同一平台同一创作者不能重复订阅
    __table_args__ = (
        UniqueConstraint('platform', 'creator_id', name='uq_platform_creator'),
    )
```

### 2.2 FieldMappingScheme + FieldMappingItem — 字段映射

```python
class FieldMappingScheme(Base):
    """字段映射方案 (一组映射规则)"""
    __tablename__ = "webui_field_mapping_scheme"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    name            = Column(String(128), nullable=False)
    # 方案名称: "默认方案", "飞书完整版", "CSV精简版"

    platform        = Column(String(20), nullable=False, index=True)
    # 平台: xhs | wechat | dy | ...

    data_type       = Column(String(20), nullable=False)
    # 数据类型: note | comment | creator | article

    description     = Column(Text, default="")
    # 方案描述

    is_default      = Column(Boolean, default=False)
    # 是否为该平台+数据类型的默认方案

    is_system       = Column(Boolean, default=False)
    # 是否为系统内置方案 (不可删除)

    created_at      = Column(DateTime, default=func.now())
    updated_at      = Column(DateTime, default=func.now(), onupdate=func.now())

    # 关系
    items           = relationship("FieldMappingItem", back_populates="scheme",
                                   cascade="all, delete-orphan",
                                   order_by="FieldMappingItem.sort_order")

    __table_args__ = (
        UniqueConstraint('platform', 'data_type', 'name',
                         name='uq_platform_type_name'),
    )


class FieldMappingItem(Base):
    """单个字段的映射配置"""
    __tablename__ = "webui_field_mapping_item"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    scheme_id       = Column(Integer, ForeignKey("webui_field_mapping_scheme.id"),
                             nullable=False, index=True)

    source_field    = Column(String(128), nullable=False)
    # 原始字段名: "note_id", "liked_count", "title"

    display_name    = Column(String(128), nullable=False)
    # 显示名称 (中文): "笔记ID", "点赞数", "标题"

    enabled         = Column(Boolean, default=True)
    # 是否启用 (是否参与导出/同步)

    sort_order      = Column(Integer, default=0)
    # 排序序号

    feishu_type     = Column(String(20), default="text")
    # 飞书列类型: text | number | date | single_select | multi_select |
    #             url | attachment | checkbox | phone | email

    feishu_options  = Column(JSON, default=dict)
    # 飞书列额外配置:
    # - single_select: {"options": ["图文", "视频", "音频"]}
    # - number: {"formatter": "0,0"}  (千分位)
    # - date: {"date_format": "YYYY-MM-DD"}

    transform       = Column(String(50), default="none")
    # 值转换规则: none | timestamp_to_date | json_parse | truncate | ...

    transform_config = Column(JSON, default=dict)
    # 转换配置:
    # - truncate: {"max_length": 200}
    # - timestamp_to_date: {"format": "%Y-%m-%d %H:%M"}
    # - json_parse: {"extract_field": "name"}

    # 关系
    scheme          = relationship("FieldMappingScheme", back_populates="items")
```

**转换规则枚举:**

| transform | 说明 | transform_config |
|-----------|------|------------------|
| `none` | 原样输出 | — |
| `timestamp_to_date` | Unix 时间戳 → 日期字符串 | `{"format": "%Y-%m-%d"}` |
| `json_parse` | JSON 字符串 → 提取字段 | `{"extract_field": "name"}` |
| `json_to_list` | JSON 数组 → 逗号分隔文本 | `{"separator": ", "}` |
| `truncate` | 截断长文本 | `{"max_length": 200}` |
| `number_format` | 数字千分位格式化 | `{"decimals": 0}` |
| `url_prefix` | 添加 URL 前缀 | `{"prefix": "https://www.xiaohongshu.com/explore/"}` |
| `map_value` | 值映射 | `{"mapping": {"0": "图文", "5": "视频"}}` |

### 2.3 ScheduledTask + TaskExecution — 任务调度

```python
class ScheduledTask(Base):
    """定时任务定义"""
    __tablename__ = "webui_scheduled_task"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    name            = Column(String(128), nullable=False, unique=True)
    # 任务名称

    task_type       = Column(String(20), nullable=False)
    # 任务类型: crawl | sync | cleanup | combo (采集+同步)

    platform        = Column(String(20), nullable=True)
    # 关联平台 (cleanup 类型可为 null)

    is_active       = Column(Boolean, default=True)
    # 是否启用

    schedule_type   = Column(String(20), nullable=False)
    # 调度类型: interval | cron | once

    schedule_config = Column(JSON, nullable=False)
    # 调度配置:
    # interval: {"hours": 6}  或  {"minutes": 30}
    # cron:     {"hour": "8", "minute": "0", "day_of_week": "mon-fri"}
    # once:     {"run_date": "2026-02-15 08:00:00"}

    task_config     = Column(JSON, nullable=False, default=dict)
    # 任务执行配置 (因 task_type 而异)

    # crawl 配置:
    # {
    #   "crawler_type": "creator",        // search | detail | creator
    #   "login_type": "cookie",
    #   "keywords": "设计, UI",           // search 模式
    #   "subscription_ids": [1, 2, 3],    // creator 模式: 关联订阅
    #   "creator_ids": "user1,user2",     // creator 模式: 直接指定
    #   "start_page": 1,
    #   "max_pages": 5,
    #   "enable_comments": true,
    #   "save_option": "db"
    # }

    # sync 配置:
    # {
    #   "data_source": "xhs_note",        // 数据源
    #   "mapping_scheme_id": 1,           // 字段映射方案ID
    #   "date_range_type": "since_last",  // since_last | fixed | all
    #   "date_range": {"start": "2026-01-01", "end": "2026-02-11"},
    #   "target_table": "auto",           // auto (自动创建) | 指定table_id
    #   "batch_size": 500
    # }

    # combo 配置:
    # {
    #   "crawl": { ... },                 // 同 crawl 配置
    #   "sync": { ... },                  // 同 sync 配置
    #   "sync_delay_seconds": 10          // 采集完成后等待 N 秒再同步
    # }

    # cleanup 配置:
    # {
    #   "cleanup_type": "logs",           // logs | data | temp
    #   "retain_days": 30
    # }

    last_run_at     = Column(DateTime, nullable=True)
    next_run_at     = Column(DateTime, nullable=True)
    run_count       = Column(Integer, default=0)
    fail_count      = Column(Integer, default=0)

    created_at      = Column(DateTime, default=func.now())
    updated_at      = Column(DateTime, default=func.now(), onupdate=func.now())

    # 关系
    executions      = relationship("TaskExecution", back_populates="task",
                                   order_by="TaskExecution.started_at.desc()")


class TaskExecution(Base):
    """任务执行记录"""
    __tablename__ = "webui_task_execution"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    task_id         = Column(Integer, ForeignKey("webui_scheduled_task.id"),
                             nullable=False, index=True)
    task_name       = Column(String(128), nullable=False)
    # 冗余存储任务名 (即使任务删除也能看到历史)

    status          = Column(String(20), nullable=False, default="running")
    # running | success | failed | cancelled | timeout

    trigger_type    = Column(String(20), default="scheduled")
    # scheduled | manual

    started_at      = Column(DateTime, default=func.now())
    finished_at     = Column(DateTime, nullable=True)
    duration_seconds = Column(Float, nullable=True)

    # 执行结果
    result_summary  = Column(JSON, default=dict)
    # {
    #   "crawl": {"total": 47, "new": 23, "updated": 24, "errors": 0},
    #   "sync":  {"total": 47, "success": 47, "failed": 0, "table_url": "..."}
    # }

    error_message   = Column(Text, nullable=True)
    log_output      = Column(Text, nullable=True)
    # 执行日志 (最近 N 行)

    # 关系
    task            = relationship("ScheduledTask", back_populates="executions")
```

```python
class SyncHistory(Base):
    """飞书同步历史记录"""
    __tablename__ = "webui_sync_history"

    # 结果统计
    total_records   = Column(Integer, default=0)
    success_count   = Column(Integer, default=0)
    failed_count    = Column(Integer, default=0)
    skipped_count   = Column(Integer, default=0)

    # 飞书目标
    feishu_app_token = Column(String(128), default="")
    feishu_table_id  = Column(String(128), default="")
    feishu_table_name = Column(String(256), default="")
    feishu_table_url  = Column(String(512), default="")
    # 飞书多维表格链接

    status          = Column(String(20), default="running")
    # running | success | partial | failed

    error_message   = Column(Text, nullable=True)

    started_at      = Column(DateTime, default=func.now())
    finished_at     = Column(DateTime, nullable=True)
    duration_seconds = Column(Float, nullable=True)
```

## 5. 数据库迁移策略

### 5.1 原则

- WebUI 新增表使用 `webui_` 前缀，与已有爬虫数据表完全隔离
- **不修改** `database/models.py`，WebUI 模型放在独立的 `database/webui_models.py`
- 共享同一个 `Base` 实例，利用现有 `create_all()` 自动建表 —— **不需要 Alembic**
- WebUI 以只读方式访问已有爬虫数据表

### 5.2 webui_models.py 完整头部

```python
# database/webui_models.py  (🆕 新建文件)

WebUI 专用 ORM 模型。
共享 database.models.Base，所有表名以 webui_ 开头。
"""

    Column, Integer, String, Text, Boolean, Float, DateTime, JSON,
    ForeignKey, UniqueConstraint, func
)
```

### 5.3 现有文件改动 (仅 1 行)

```python

async def init_db(db_type: str = None):
    import database.webui_models  # 🆕 确保 WebUI 表注册到 Base.metadata
    await init_table_schema(db_type)
```

当 `init_table_schema()` 内部调用 `Base.metadata.create_all()` 时，
由于 `webui_models.py` 中的 7 个类已经注册到同一个 `Base.metadata`，
`create_all()` 会自动创建对应的 `webui_*` 表 —— **幂等操作，不影响已有表**。

### 5.4 种子数据初始化

种子数据在 FastAPI startup 事件中注入，代码位于新建的 `api/services/webui_init.py`：

```python
# api/services/webui_init.py  (🆕 新建文件)
from database.db_session import get_session
from database.webui_models import FieldMappingScheme, FieldMappingItem

# 默认映射定义 (结构见 Section 3)
DEFAULT_MAPPINGS = { ... }

async def seed_default_mappings():
    """插入默认字段映射方案 (仅首次运行, 幂等)"""
    async with get_session() as session:
        if session is None:
            return  # 当前存储模式为 CSV/JSON, 跳过

        existing = await session.execute(
            select(FieldMappingScheme).where(FieldMappingScheme.is_system == True)
        )
        if existing.scalars().first():
            return  # 已初始化

        for platform_key, mappings in DEFAULT_MAPPINGS.items():
            scheme = FieldMappingScheme(
                name="系统默认",
                platform=platform_key["platform"],
                data_type=platform_key["data_type"],
                is_default=True,
                is_system=True,
            )
            session.add(scheme)
            await session.flush()

            for i, field in enumerate(mappings):
                session.add(FieldMappingItem(
                    scheme_id=scheme.id,
                    source_field=field["source"],
                    display_name=field["display"],
                    enabled=field["enabled"],
                    sort_order=i + 1,
                    feishu_type=field["feishu_type"],
                ))
        await session.commit()
```

在 `api/main.py` 的 startup 事件中调用：

```python
# api/main.py  (✏️ 修改 — 在 startup 事件中追加)
from api.services.webui_init import seed_default_mappings

@app.on_event("startup")
async def on_startup():
    # ... 已有初始化逻辑 ...
    await seed_default_mappings()  # 🆕
```

### 5.5 会话复用保证

所有 WebUI 服务 **不创建新引擎**，统一使用 `database.db_session` 提供的设施：

| 设施 | 来源 | 说明 |
|------|------|------|
| `get_async_engine()` | `database/db_session.py` | 引擎工厂 (内部有缓存) |
| `get_session()` | `database/db_session.py` | 异步 Session 上下文管理器 |
| `Base` | `database/models.py` | 声明式基类 |

这意味着：
- WebUI 和爬虫系统使用 **同一个数据库连接池**
- 数据库类型 (SQLite / MySQL / PostgreSQL) 由 `.env` 中的 `SAVE_DATA_OPTION` 决定
- 若用户使用 CSV/JSON 模式，`get_session()` 返回 `None`，WebUI 功能降级为只读配置管理
